cmake_minimum_required(VERSION 3.12)
project(ORGraph)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Для Windows
if(WIN32)
    add_definitions(-DNOMINMAX)
    set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)  # Экспортируем все символы для Windows
endif()

# 1. Находим Python
find_package(Python3 REQUIRED COMPONENTS Development Interpreter)
message(STATUS "Python found: ${Python3_VERSION}")

# 2. Ищем pybind11 через Python
execute_process(
    COMMAND ${Python3_EXECUTABLE} -c "
try:
    import pybind11
    import os
    print(os.path.join(os.path.dirname(pybind11.__file__), 'include'))
except ImportError:
    # Пробуем установить
    import subprocess
    import sys
    subprocess.check_call([sys.executable, '-m', 'pip', 'install', 'pybind11'])
    import pybind11
    import os
    print(os.path.join(os.path.dirname(pybind11.__file__), 'include'))"
    OUTPUT_VARIABLE PYBIND11_INCLUDE_DIR
    OUTPUT_STRIP_TRAILING_WHITESPACE
)

message(STATUS "pybind11 include: ${PYBIND11_INCLUDE_DIR}")

# 3. Проверяем nlohmann_json (опционально)
set(HAVE_NLOHMANN_JSON FALSE)
if(EXISTS "${CMAKE_SOURCE_DIR}/include/nlohmann/json.hpp" OR EXISTS "${CMAKE_SOURCE_DIR}/nlohmann/json.hpp")
    set(HAVE_NLOHMANN_JSON TRUE)
    message(STATUS "Found nlohmann_json in project")
endif()

# Include directories
include_directories(
    ${PROJECT_SOURCE_DIR}/ORGraph/include
    ${PROJECT_SOURCE_DIR}/ORGraph
    ${Python3_INCLUDE_DIRS}
    ${PYBIND11_INCLUDE_DIR}
)

# Source files
set(SOURCES
    ORGraph/src/Graph.cpp
    ORGraph/orgraph_bindings.cpp
    ORGraph/src/algorithms/deijkstra_algorithm.cpp
    ORGraph/src/algorithms/prim_algorithm.cpp
    ORGraph/src/algorithms/ford_bellman_algorithm.cpp
    ORGraph/src/algorithms/floyd_algorithm.cpp
    ORGraph/src/algorithms/euler_path_algorithm.cpp
    ORGraph/src/algorithms/hamiltonian_path_algorithm.cpp
    ORGraph/src/algorithms/topology_sort.cpp
    ORGraph/src/GraphGenerator.cpp
    ORGraph/src/representations/AdjacencyList.cpp
    ORGraph/src/representations/AdjacencyMatrix.cpp
)

# Build python module (без pybind11_add_module)
add_library(orgraph_core SHARED ${SOURCES})

# Для Windows: создаем .pyd файл
if(WIN32)
    set_target_properties(orgraph_core PROPERTIES
        SUFFIX ".pyd"
        OUTPUT_NAME "orgraph_core"
        PREFIX ""
    )
else()
    # Для Linux
    set_target_properties(orgraph_core PROPERTIES
        SUFFIX ".so"
        OUTPUT_NAME "orgraph_core"
        PREFIX ""
    )
endif()

# Link libraries
target_link_libraries(orgraph_core PRIVATE ${Python3_LIBRARIES})

# Output directory
set_target_properties(orgraph_core PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/web_interface/server"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/web_interface/server"
)

message(STATUS "Output directory: ${CMAKE_SOURCE_DIR}/web_interface/server")
message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")
